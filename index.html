Petunjuk Format Kode ESP32 Arduino (Web Server + CORS)
Agar browser dapat membaca respons dari ESP32 tanpa terhalang Cross-Origin Resource Sharing (CORS), pastikan kode ESP32 menyertakan header Access-Control-Allow-Origin: * seperti contoh berikut: 
#include <WiFi.h>
#include <WebServer.h>

WebServer server(80);
const int SOIL_PIN = 34; // Pin ADC ESP32

void handleRoot() {
  int rawValue = analogRead(SOIL_PIN);
  
  // Wajib tambahkan header CORS ini agar javascript fetch() dari browser berhasil:
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "*");
  
  // Kirim payload JSON sesuai format
  String json = "{\"soil_moisture\":" + String(rawValue) + "}";
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);
  WiFi.begin("SSID_HOTSPOT", "PASSWORD");
  // Tunggu koneksi wifi...
  server.on("/", handleRoot);
  server.begin();
  Serial.print("ESP32 IP: ");
  Serial.println(WiFi.localIP());
}
